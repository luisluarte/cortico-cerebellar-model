	\begin{frame}{What is the actual problem?}
		\begin{itemize}
			\item With reversal-learning task lets see if we can derive a value and uncertainty signal
			\item What, in more general terms, is a reversal-learning task?
			\item If option A $\rightarrow$ $+0$
			\item Does the $(\text{Action, Reward})$ tuple at time $t$ provide enough information to determine if $\text{option A} < \text{option B}$
			\item Agent must know the last $N$ trials to determine $A > B \ | \ A < B$
			\item Agent wants to determine $(\text{Reward} \ | \ \pi)$, so it must derive $\text{Value}(\pi) \ | \ f(O_{t}, O_{t-1}, \dots, O_{t - N})$, where $O_{t}$ is the observed reward and environment at time $t$
			\item This gives us a solid frame-work to start with, formally this same problem is partially observable markov decision process (POMDP)
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Reversal-learning is a non-markovian environment for a limited information agent}
		\begin{itemize}
			\item We know that in a POMDP there's a state $S_{t}$ which transition to $S_{t+1}$ is markovian, thus depends only on $S_{t}$
			\item However, we also know that from the perspective of a limited information agent, the environment behaves non-markovian as $O_{t} \rightarrow O_{t+1}$ depends on $S$ which is unknown and presents as a latent state
			\item Ideal agent needs to derive $S = f(O_{t}, \dots)$, in other words, the agent needs to know if its winter or autumm by checking if its raining or not
		\end{itemize}
		
		\begin{block}{TLDR}
			\begin{itemize}
				\item Reversal learning $\rightarrow$ non-markovian
				\item non-markovian $\rightarrow$ requires temporal integration
				\item The integration of $S, R, V(\pi)$ $\rightarrow$ Elegibility trace
			\end{itemize}
		\end{block}
	\end{frame}

\begin{frame}{The gap in canonical models}
    \begin{itemize}
    	\item \textbf{Fading memory.} how does the granule layer buffer historical states across non-markovian temporal (or other) gaps
    	\item \textbf{Cortico-Cerebellar feedback loop.} canonical models heavily favored a \textit{feedforward} mode in the cerebellum (though as pure motor controller). Largely ignoring or not explicitly modeling the projections from DCN to the CTX via Thal
    	\item \textbf{Credit assignment.} canonical models relied on instantaneous motor errors. Lacking a mechanism to map a delayed, abstract reward signal back onto a sequence of past hidden states, to overcome cognitive policy perserveration
    \end{itemize}
    
    \vspace{4mm}
    \begin{block}{The Goal of Our Model}
        To mathematically formulate and simulate these missing recurrent and temporal dynamics, proving the cerebellum's capacity to resolve non-Markovian environments. Then if we capture a value and uncertainty signal it would be a more 'emergent' less bias finding.
    \end{block}
\end{frame}

		\begin{frame}{Cortex hallucination of sensory input}
		\begin{equation}
			\hat{\mathbf{I}}_t = \mathbf{W}_{gen} \boldsymbol{\mu}_{t}
		\end{equation}
		\begin{itemize}
			\item $\hat{\mathbf{I}}$, is the $\textit{hallucination}$ of equal dimension to $\mathbf{I}$ $\textit{actual input}$
			\item $\mathbf{W}_{gen}$ is the generative matrix than makes the lower-dimension projection, intialized random and fixed $\mathcal{N}(0, \nicefrac{1}{\sqrt{\dim{\boldsymbol{\mu}}}})$
			\item $\boldsymbol{\mu}_{t}$ is the $\textit{cortical state}$ at time $t$
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Cortical prediction error}
		\begin{equation}
			\boldsymbol{\epsilon}_{t} = \Pi \odot (\mathbf{I}_{t} - \hat{\mathbf{I}}_{t})
		\end{equation}
		\begin{itemize}
			\item The cortex makes a $\textit{hallucination}$ which acts as active inference of the world, then the participant makes an actual choice generating the sensory vector $\mathbf{I}_{t}$
			\item To build $\Pi$ as as the precision weighting, we choose the most simplest one. Static anatomical precision, let $\Pi_{raw, i} = \nicefrac{1}{\sigma(\mathbf{I}_{i})}$ be the raw precision weighting vector for input $i$ with variance $\sigma(\mathbf{I}_{i})$, then we generate $\Pi = \nicefrac{\Pi_{raw}}{\bar{\Pi}_{raw}}$ to be the normalized precision weighting
			\item Element-wise multiplication means we build prediction error $\boldsymbol{\epsilon}_t$ as the precision weighted prediction error
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Propagation of cortical state in to the Cerebellum}
		\begin{equation}
			\mathbf{G}_{t} = \mathbf{W}_{ach1} \boldsymbol{\mu}_{t}
		\end{equation}
		\begin{itemize}
			\item $\mathbf{G}_{t}$ is the Granule Layer state at time $t$, defined as a Marr-Albus-Ito random sparse projection
			\item We modeled this as an Achlioptas random sparse projection ($896 \times 32$), where each weight $w_{i, j}$ is assigned with probability $p$: $-1$ or $1$, and 0 with $1 - 2p$, setting sparisity ($1 - 2p$) to 90\% zeros
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Cebellum memory or eligibility trace}
		When the cortex makes a prediction and initiates an action, there is some delay for this feedback to arrive. How do we solve the problem of pairing some particular $\mathbf{G}_{t}$ with a given feedback. We need to leave an $\textit{eligibility trace}$
		\begin{equation}
			\mathbf{Z}_{t} = (1 - \beta_{gran}) \mathbf{Z}_{t-1} + \alpha_{gran} \mathbf{G}_{t}
		\end{equation}
		\begin{itemize}
			\item Where $\mathbf{G}_{t}$ is the instantaneous firing of the Granule Layer
			\item $\alpha_{gran}$ is the parameter that controls how aggresively the layer updates
			\item $(1 - \beta_{gran})$ is the $\textit{leaky bucket}$, that decays the trace on every trial
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Cerebellum learning}
		To learn the Cerebellum projects the 6-Dimensional precision weighted error $\boldsymbol{\epsilon}_t$ to a single dimension (relating this to the Inferior Olive)
		\begin{equation}
			\left| \bar{\epsilon}_{t} \right| = \frac{1}{d} \sum_{i=1}^{d} \left| \boldsymbol{\epsilon}_{t, i} \right|
		\end{equation}
		With that scalar error we update the Granule Layer trace in its Purkinje Cell representation via anti-Hebbian learning
		\begin{equation}
			\mathbf{W}_{purk, t} = \mathbf{W}_{purk, t-1} - \kappa ( \left| \bar{\epsilon}_{t} \right| \mathbf{Z}_{t})
		\end{equation}
	\end{frame}
	
	\begin{frame}{Cerebellum compression}
		\begin{equation}
			\mathbf{D}_{t} = \mathbf{W}_{ach2} (\mathbf{G}_{t} \odot \mathbf{W}_{purk, t})
		\end{equation}
		\begin{itemize}
			\item $\mathbf{D}_{t} \in \mathbb{R}^{362}$  is the Cerebellum forward pass
			\item $\mathbf{G}_{t} \odot \mathbf{W}_{purk, t}$ is the element-wise multiplication between the Granule Layer current state and the LTD Purkinje layer we previously constructed
			\item Finally with $\mathbf{W}_{purk, t}$ we perform an random sparse matrix compression into the 362 dimension of the cerebellar forward pass
		\end{itemize}
	\end{frame}
	
	\begin{frame}{The loop-back: thalamo-cortical pass}
		\begin{equation}
			\boldsymbol{\mu}_{t+1} = \boldsymbol{\mu}_{t} + \alpha_{c} \left[(\mathbf{W}^{T}_{gen} \boldsymbol{\epsilon}_{t}) - (\lambda_{c} \cdot ITI_{t} \cdot \boldsymbol{\mu}_{t}) + \beta_{thal} (\mathbf{W}_{thal} \mathbf{D}_{t})\right]
		\end{equation}
		\begin{itemize}
			\item This is the Cortical trace update, $\mathbf{W}^{T}_{gen} \boldsymbol{\epsilon}_{t}$ projects the error back into the cortical dimension $6 \rightarrow 32$, and we perform gradient descent over a time-decaying trace $\lambda_{c} \cdot ITI_{t} \cdot \boldsymbol{\mu}_{t}$
			\item $\mathbf{W}_{thal}$ is the connectivity from cerebellum to cortex $362 \rightarrow 32$, so $\mathbf{W}_{thal} \cdot \mathbf{D}_{t}$ is the cerebellum-cortex forward pass, which via $\beta_{thal}$ pushes the cortex into the non-linear projection
			\item Finally, the trace is updated via learning rate $\alpha_{c}$ 
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Converting a trace into action and reaction-time}
		For empirical prediction, $M_{cc}$ utilizes an L2-regularized Ridge regression readout to map the cortical latent state $\boldsymbol{\mu}_t$ onto the observed behavioral variables (binary choice and continuous reaction time):
		\begin{equation}
			P(\text{Choice}_t = 1 \mid \boldsymbol{\mu}_t) = \sigma(\mathbf{w}_{\text{choice}}^T \boldsymbol{\mu}_t)
		\end{equation}
		\begin{equation}
			RT_t \sim \mathcal{N}(\mathbf{w}_{\text{rt}}^T \boldsymbol{\mu}_t, \sigma_{\text{rt}}^2)
		\end{equation}
		where $\sigma(\cdot)$ denotes the logistic sigmoid function, and $\mathbf{w}_{\text{choice}}$ and $\mathbf{w}_{\text{rt}}$ are the respective optimal linear readout weights.
	\end{frame}




\begin{frame}{Methods: Teacher-Student Distillation}
    \begin{itemize}
        \setlength\itemsep{1.5em}
        \item \textbf{The Teacher Proxy ($RNN_{teacher}$):} We trained a Gated Recurrent Unit (GRU) to predict both a binary choice policy (BCE) and a continuous reaction time kinematic head (Log-Normal mixture). 
        \item \textbf{Calibration:} The RNN's raw logits were Platt-scaled to produce true probability distributions, scored via Continuous Ranked Probability Score (CRPS).
        \item \textbf{The Distillation Loss ($\mathcal{L}_{Distill}$):} $M_{cc}$ continuously projected its cortical state ($\boldsymbol{\mu}_t^{bio}$) to match the Teacher's latent state ($\boldsymbol{\mu}_t^{RNN}$). This was formalized as a robust Gaussian likelihood over the residual space:
        \begin{equation}
        \mathcal{L}_{Distill} = \sum_{t} \left[ -\frac{1}{2}\ln(2\pi\sigma_{res}^2) - \frac{(\boldsymbol{\mu}_t^{RNN} - \mathbf{W}_{proj}\boldsymbol{\mu}_t^{bio})^2}{2\sigma_{res}^2} \right]
        \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{Methods: Bayesian Inference \& Generalization (Math \& Intuition)}
    \begin{itemize}
        \item \textbf{Adaptive Metropolis-Hastings (MCMC):} To fit the hierarchical behavioral models, we utilized a custom C++ Adaptive Random Walk Metropolis-Hastings algorithm (Haario et al., 2001). \textbf{Intuition:} The sampler iteratively proposes parameter updates using an empirically adapted covariance matrix, dynamically scaling the step size via a Robbins-Monro stochastic approximation to maintain an optimal 23.4\% acceptance rate.
        \item \textbf{Leave-One-Out Cross Validation:} We evaluated out-of-sample generalization using Pareto-Smoothed Importance Sampling (PSIS-LOO) (Vehtari et al., 2017). 
        \item \textbf{Expected Log Predictive Density (ELPD):} PSIS-LOO estimates how well the model predicts held-out data. 
        \begin{equation}
        \text{ELPD} = \sum_{i=1}^N \log \int p(y_i | \theta) p(\theta | y_{-i}) d\theta
        \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{Methods: Quantifying Flow \& Encoding Capacity (Math \& Intuition)}
    \begin{itemize}
        \item \textbf{Granger Causality ($F$):} Does the past temporal sequence of the Cerebellum's state help predict the future of the Cortex's state better than the Cortex's own past alone? Higher $F$-values indicate stronger directional information flow.
        \item \textbf{Raw Encoding Capacity ($R^2$):} L2-regularized Ridge regression to decode the latent cognitive variable ($y_t$) solely from the region's continuous state vector ($\mathbf{X}_t$).
        \begin{equation}
        R^2 = 1 - \frac{\sum (y_t - \mathbf{X}_t \mathbf{w})^2}{\sum (y_t - \bar{y})^2}
        \end{equation}
        \item \textbf{Stabilized Readout Efficiency:} We penalized the raw predictive power by the intrinsic temporal volatility (trajectory derivative) of the region's state space:
        \begin{equation}
        \text{Efficiency} = \frac{R^2}{\text{Volatility}} = \frac{R^2}{\mathbb{E}[||\mathbf{X}_{t} - \mathbf{X}_{t-1}||^2]}
        \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{Results: Directional Flow \& Raw Encoding (Statistics)}

        \item \textbf{Stabilized Efficiency:} The Cortex demonstrated significantly higher stabilized readout efficiency (Value $\Delta=1.73$, State $\Delta=0.89$, Volatility $\Delta=1.41$, all $p<.0001$), acting as a temporal stabilizer against rapid motor errors.
        \item \textbf{Raw Encoding Capacity:} However, the Cerebellum matched or exceeded the Cortex in raw, unpenalized encoding capacity for Value ($p=.241$) and Volatility ($p=.768$), and was significantly higher for State ($p<.0001$).
    \end{itemize}

\begin{frame}{Methods: Quantifying Policy Perseverance (Math \& Intuition)}
    \begin{itemize}
        \item \textbf{Heuristic Definition:} We defined a Win-Stay/Lose-Shift policy event. If a participant received a reward and repeated the action, it is a "Win-Stay" ($w$). If they missed and switched, it is a "Lose-Shift" ($l$).
        \item \textbf{Bayesian Update:} Instead of a static ratio, we modeled their continuous policy state on a trial-by-trial basis using a conjugate Beta posterior:
        \begin{equation}
        X \sim \text{Beta}(\alpha_0 + w, \beta_0 + l)
        \end{equation}
        where $\alpha_0=\beta_0=1$ acts as a uniform prior.
        \item \textbf{Differential Entropy ($h(X)$):} We computed the differential entropy of this continuous Beta distribution. \textbf{Intuition:} It measures the "width" or uncertainty of the policy. If the network is trapped in a rigid heuristic (e.g., $100\%$ Win-Stay), the distribution spikes and entropy collapses.
    \end{itemize}
\end{frame}

\begin{frame}{Discussion}
    \begin{itemize}
        \setlength\itemsep{1.5em}
        \item \textbf{The Core Finding:} The cortico-cerebellar loop is strictly required to buffer historical action trajectories and overcome behavioral perseveration in non-Markovian environments.
        \item \textbf{A Topological Necessity:} Bayesian estimation confirms that unconstrained recurrent states overfit; tight topological bounds (like the granular expansion and Purkinje compression) are required for reliable, out-of-sample decision making.
        \item \textbf{Future Directions:} Future models should leverage direct physiological data to inform Bayesian priors, grounding the synthetic hyperparameters ($\alpha_c, \kappa, \beta_{thal}$) directly in in-vivo cerebellar firing rates and temporal dynamics.
    \end{itemize}
